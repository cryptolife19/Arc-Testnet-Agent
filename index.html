MetaMask (Clean & Direct Code) document.getElementById("connectMetaMask").onclick = async () => { modal.style.display = "none"; let provider = window.ethereum; // Multiple Providers Filter Logic if (window.ethereum?.providers) { provider = window.ethereum.providers.find(p => p.isMetaMask && !p.isPhantom && !p.isBitget); } if (!provider || (!provider.isMetaMask && !window.ethereum)) { alert("MetaMask extension is not installed!"); return; } try { currentActiveProvider = provider; // Direct Request to Open MetaMask Popup const accounts = await provider.request({ method: 'eth_requestAccounts' }); if (!accounts || accounts.length === 0) { alert("Please unlock your MetaMask wallet and select an account."); return; } const chainId = await provider.request({ method: 'eth_chainId' }); updateUIConnected(accounts[0], parseInt(chainId, 16)); } catch (err) { if (err.code === -32002) { alert("MetaMask is already open. Please check your extension popup!"); } else { alert("Connection failed: " + err.message); } } };
